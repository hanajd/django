% 单页横置 A3（仅影响夹在 \BeginAThreeLandscapePage … \EndAThreeLandscapePage 之间的内容）
% 页边距与正文 A4 一致：上下 26mm、左右 28mm
% 注意：geometry 的 \newgeometry 会忽略 paperwidth/paperheight，PDF 纸张用
% \erpSetPDFPageSize 切换后，必须再显式把 \textwidth/\headwidth 拉到 A3 版心。
\newcommand{\erpSetPDFPageSize}[2]{%
  \setlength{\paperwidth}{#1}%
  \setlength{\paperheight}{#2}%
  \ifPDFTeX
    \pdfpagewidth=#1\relax
    \pdfpageheight=#2\relax
  \fi
  \ifXeTeX
    \pdfpagewidth=#1\relax
    \pdfpageheight=#2\relax
  \fi
  \ifLuaTeX
    \pagewidth=#1\relax
    \pageheight=#2\relax
  \fi
}
\newcommand{\erpApplyAThreeLandscapeLayout}{%
  % geometry 的 \newgeometry 不会改纸张宽度，版心会仍按 A4；此处强制 A3 横置版心
  \setlength{\textwidth}{364mm}%
  \setlength{\linewidth}{\textwidth}%
  \setlength{\columnwidth}{\textwidth}%
  \setlength{\hsize}{\textwidth}%
  \setlength{\headwidth}{\textwidth}%
  \setlength{\oddsidemargin}{\dimexpr 28mm-1in\relax}%
  \setlength{\evensidemargin}{\oddsidemargin}%
  % 高度与 A4 相同（297mm），沿用 newgeometry 的垂直参数即可
  \fancyhfoffset{0pt}%
}
\newcommand{\BeginAThreeLandscapePage}{%
  \clearpage
  \erpSetPDFPageSize{420mm}{297mm}%
  \newgeometry{%
    top=26mm,%
    bottom=26mm,%
    left=28mm,%
    right=28mm,%
    headheight=15pt,%
    includeheadfoot%
  }%
  \erpApplyAThreeLandscapeLayout
  \thispagestyle{fancy}%
}
\newcommand{\EndAThreeLandscapePage}{%
  \clearpage
  \erpSetPDFPageSize{210mm}{297mm}%
  \newgeometry{%
    top=26mm,%
    bottom=26mm,%
    left=28mm,%
    right=28mm,%
    headheight=15pt,%
    includeheadfoot%
  }%
  \setlength{\headwidth}{\textwidth}%
  \fancyhfoffset{0pt}%
}
